CREATE TABLE books (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    author VARCHAR(255),
    genre VARCHAR(100),
    price DECIMAL(6,2),
    published_year INT,
    in_stock BOOLEAN DEFAULT TRUE
);

## Answer Key

<details>
<summary>Reveal answers</summary>

```sql title="answers.sql"
-- 1. Total number of books
SELECT COUNT(*) FROM books;

-- 2. Total value of all books
SELECT SUM(price) FROM books;

-- 3. Average price of all books
SELECT AVG(price) FROM books;

-- 4. Cheapest book's price
SELECT MIN(price) FROM books;

-- 5. Most expensive book's price
SELECT MAX(price) FROM books;

-- 6. Count of in-stock books
SELECT COUNT(*) FROM books WHERE in_stock = TRUE;

-- 7. Number of books per genre
SELECT genre, COUNT(*) AS book_count
FROM books
GROUP BY genre;

-- 8. Average price per author
SELECT author, AVG(price) AS avg_price
FROM books
GROUP BY author;

-- 9. Total price per genre
SELECT genre, SUM(price) AS total_price
FROM books
GROUP BY genre;

-- 10. Earliest and latest published_year per author
SELECT author, MIN(published_year) AS earliest, MAX(published_year) AS latest
FROM books
GROUP BY author;

-- 11. Genres with more than 2 books
SELECT genre, COUNT(*) AS book_count
FROM books
GROUP BY genre
HAVING COUNT(*) > 2;

-- 12. Authors with average price above $12
SELECT author, AVG(price) AS avg_price
FROM books
GROUP BY author
HAVING AVG(price) > 12;

-- 13. In-stock books per genre (filter rows before grouping)
SELECT genre, COUNT(*) AS book_count
FROM books
WHERE in_stock = TRUE
GROUP BY genre;

-- 14. Genres where total combined price exceeds $20
SELECT genre, SUM(price) AS total_price
FROM books
GROUP BY genre
HAVING SUM(price) > 20;

-- 15. Fails because "price" is neither aggregated nor in GROUP BY —
-- MySQL doesn't know which row's price to show per genre group.
-- Fix: either aggregate it, or remove it from SELECT.
SELECT genre, AVG(price), COUNT(*)
FROM books
GROUP BY genre;

-- 16. Fails because WHERE runs before grouping happens, so COUNT(*)
-- doesn't exist yet at that point. HAVING must be used instead.
SELECT author, COUNT(*)
FROM books
GROUP BY author
HAVING COUNT(*) > 1;

-- 17. Genres averaging above $10 AND with at least 2 books
SELECT genre, AVG(price) AS avg_price, COUNT(*) AS book_count
FROM books
GROUP BY genre
HAVING AVG(price) > 10 AND COUNT(*) >= 2;

-- 18. Author with the highest total combined book value
SELECT author, SUM(price) AS total_value
FROM books
GROUP BY author
ORDER BY total_value DESC
LIMIT 1;
```
